@echo off
chcp 65001 >nul
echo [pack] version=%VERSION% tag=%TAG_NAME%
call pnpm install --frozen-lockfile
if errorlevel 1 exit /b 1
call pnpm run build
if errorlevel 1 exit /b 1
xcopy /e /i /y "%OUTPUT_PATH%" "%ARTIFACTS_DIR%\%OUTPUT_PATH%" >nul
if errorlevel 1 exit /b 1
echo [pack] done

@echo off
chcp 65001 >nul

echo [pack] version=%VERSION% tag=%TAG_NAME%

call pnpm install --frozen-lockfile --registry=http://192.168.1.10:8081/repository/npm-group/
if errorlevel 1 exit /b 1

call pnpm run build
if errorlevel 1 exit /b 1

REM 产物复制到平台产物目录
xcopy /e /i /y "%OUTPUT_PATH%" "%ARTIFACTS_DIR%\%OUTPUT_PATH%" >nul
if errorlevel 1 exit /b 1
echo [pack] done %VERSION%
